# Pinned to the versions the loader is known to work with locally.
# Regenerate deliberately, not opportunistically: an unpinned nba_api upgrade
# can change column names in the stats frames and silently blank out fields.
#
# 1.6.1 -> 1.11.4 was verified before pinning, by running both versions against
# the live endpoints and diffing what the loader would actually insert. Every
# column it reads still exists and still carries values, and the top of the
# league is unchanged. What does change is the bundled static player list:
# 1.6.1's predates the 2025 draft, so the loader was silently skipping that
# whole class. See verify_player_table.py for the check that now guards this.
nba_api==1.11.4

# nba_api's own pins are loose, so pin what it pulls in as well.
pandas==3.0.5
numpy==1.26.4
requests==2.34.2

# psycopg2-binary, not psycopg2: the ubuntu runner has no libpq-dev, so the
# source distribution fails to build. Same `import psycopg2` either way.
psycopg2-binary==2.9.12
